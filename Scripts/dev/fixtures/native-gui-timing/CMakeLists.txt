cmake_minimum_required(VERSION 3.24)
project(gui_timing_controls LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml)
add_library(timingclock MODULE clock.cpp)
target_link_libraries(timingclock PRIVATE Qt6::Core Qt6::Qml)
set_target_properties(timingclock PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/GuiTiming")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/GuiTiming")
file(WRITE "${CMAKE_BINARY_DIR}/GuiTiming/qmldir" "module GuiTiming\nplugin timingclock\n")
add_library(readycontrol MODULE control.cpp)
target_link_libraries(readycontrol PRIVATE Qt6::Core Qt6::Qml)
set_target_properties(readycontrol PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/ReadyControl")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/ReadyControl")
file(WRITE "${CMAKE_BINARY_DIR}/ReadyControl/qmldir" "module ReadyControl\nplugin readycontrol\n")
add_library(ready-delay SHARED delay.cpp)
target_link_libraries(ready-delay PRIVATE dl)
